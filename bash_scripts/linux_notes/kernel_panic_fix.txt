##  Remediation for kernel panic that occurred after 5.15.0-198 k update

##  Purge kernel versions ##

apt-get purge linux-image-5.15.0-198-generic
apt-get purge linux-image-5.15.0-198
apt-get purge linux-headers-5-15.0-198-generic
apt-get purge linux-headers-5-15.0-198

##  Delete kernel files under /boot ## 

rm /boot/config-5.15.0-198-generic
rm /boot/System.map-5.15.0-198-generic
rm /boot/vmlinuz-5.15.0-198-generic

##  Confirm purge ##

dpkg --list | grep linux-image-5.15.0-1*

##  Update GRUB ##

update-grub

reboot
